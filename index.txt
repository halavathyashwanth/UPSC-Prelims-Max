<script>
  // ── Config ────────────────────────────────────────────────────────────────
  // Change this to your deployed backend URL in production
  // e.g. 'https://upsc-engine-backend.onrender.com'
  const API_BASE = 'https://upsc-prelims-max.onrender.com';

  // ── Helpers ───────────────────────────────────────────────────────────────
  function updateCount() {
    const len = document.getElementById('questionInput').value.length;
    document.getElementById('charCount').textContent = len + ' / 5000 chars';
  }

  function showError(html) {
    const el = document.getElementById('errorCard');
    el.innerHTML = html;
    el.style.display = 'block';
  }

  function hideError() {
    document.getElementById('errorCard').style.display = 'none';
  }

  function toggleSection(idx) {
    const body = document.getElementById('sb-' + idx);
    const chev = document.getElementById('chev-' + idx);
    const head = document.getElementById('sh-' + idx);
    if (!body) return;
    const isOpen = body.classList.contains('open');
    body.classList.toggle('open', !isOpen);
    chev.classList.toggle('open', !isOpen);
    head.classList.toggle('open', !isOpen);
  }

  function copyFlash() {
    const txt = document.getElementById('flashcardText').textContent;
    navigator.clipboard.writeText(txt).then(() => {
      const btn = document.getElementById('copyBtn');
      btn.textContent = '✓ Copied!';
      btn.style.background = '#065f46';
      btn.style.color = '#fff';
      setTimeout(() => {
        btn.textContent = 'Copy';
        btn.style.background = '';
        btn.style.color = '';
      }, 2200);
    });
  }

  // ── Main analysis ─────────────────────────────────────────────────────────
  async function runAnalysis() {
    const q = document.getElementById('questionInput').value.trim();
    if (!q) { alert('Please paste a UPSC question first.'); return; }
    if (q.length < 10) { alert('Question seems too short. Please paste the full question.'); return; }

    hideError();
    document.getElementById('outputPanel').style.display = 'none';
    document.getElementById('loaderWrap').style.display = 'block';
    document.getElementById('analyzeBtn').disabled = true;

    try {
      const resp = await fetch(API_BASE + '/api/analyze', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ question: q })
      });

      const data = await resp.json();

      if (!resp.ok) {
        throw new Error(data.error || 'Server error ' + resp.status);
      }

      renderOutput(data.analysis);

    } catch (err) {
      const msg = err.message || 'Unknown error';
      if (msg.includes('fetch') || msg.includes('Failed') || msg.includes('NetworkError')) {
        showError(
          '<strong>⚠️ Cannot reach the backend server.</strong><br>' +
          'Make sure your backend is running at <code>' + API_BASE + '</code>. ' +
          'If you have deployed it, update the <code>API_BASE</code> variable in <code>index.html</code> to your deployed URL.'
        );
      } else {
        showError('<strong>⚠️ Analysis failed:</strong> ' + msg);
      }
    } finally {
      document.getElementById('loaderWrap').style.display = 'none';
      document.getElementById('analyzeBtn').disabled = false;
    }
  }

  // ── Render ────────────────────────────────────────────────────────────────
  function renderOutput(d) {
    // Verdict
    document.getElementById('verdictAnswer').textContent = d.answer || '—';
    document.getElementById('verdictAnchor').textContent = '📌 ' + (d.anchor || '—');

    // Risk
    const score = Math.min(5, Math.max(1, parseInt(d.riskScore) || 3));
    document.getElementById('riskFill').style.width = (score / 5 * 100) + '%';
    document.getElementById('riskScore').textContent = d.risk || (score + '/5');

    // Trap
    document.getElementById('trapBadge').innerHTML =
      '<span class="trap-badge">' + escHtml(d.trap || '—') + '</span>';

    // Build sections
    const autopsyHtml = (d.autopsy || []).map(s => `
      <div class="stmt-row ${s.correct ? 'correct' : 'wrong'}">
        <span class="stmt-icon">${s.correct ? '✅' : '❌'}</span>
        <div class="stmt-content ${s.correct ? 'correct' : 'wrong'}">
          <strong>${escHtml(s.stmt)}</strong>
          <span>${escHtml(s.explanation)}</span>
        </div>
      </div>
    `).join('');

    const sections = [
      { num: '3', title: 'Autopsy of Options',      html: autopsyHtml || '<p>—</p>' },
      { num: '4', title: 'The Elimination Drill',    html: '<p>' + escHtml(d.eliminationDrill || '—') + '</p>' },
      { num: '5', title: 'Horizontal Integration',   html: '<p>' + escHtml(d.horizontalIntegration || '—') + '</p>' },
      { num: '6', title: 'Mind Map Associations',    html: '<p>' + escHtml(d.mindMap || '—') + '</p>' },
      { num: '7', title: "Examiner's Angle",         html: '<p>' + escHtml(d.examinerAngle || '—') + '</p>' },
      { num: '8', title: 'Prediction Tag',           html: '<p>' + escHtml(d.predictionTag || '—') + '</p>' }
    ];

    document.getElementById('sectionsWrap').innerHTML = sections.map((s, i) => `
      <div class="section-card">
        <div class="section-head" id="sh-${i}" onclick="toggleSection(${i})">
          <div class="section-num">${s.num}</div>
          <div class="section-title-text">${s.title}</div>
          <span class="chevron" id="chev-${i}">▸</span>
        </div>
        <div class="section-body${i === 0 ? ' open' : ''}" id="sb-${i}">
          ${s.html}
        </div>
      </div>
    `).join('');

    // Auto-open first section head style
    const firstHead = document.getElementById('sh-0');
    if (firstHead) firstHead.classList.add('open');
    const firstChev = document.getElementById('chev-0');
    if (firstChev) firstChev.classList.add('open');

    // Flashcard
    document.getElementById('flashcardText').textContent = d.flashcard || '—';

    // Show output
    document.getElementById('outputPanel').style.display = 'block';
    document.getElementById('outputPanel').scrollIntoView({ behavior: 'smooth', block: 'start' });
  }

  function escHtml(str) {
    if (!str) return '';
    return String(str)
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
  }
